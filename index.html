package com.jarex.jarexos import android.content.Intent import android.os.Bundle import android.provider.Settings import android.view.Gravity import android.view.View import android.widget.* import android.graphics.Color import java.text.SimpleDateFormat import java.util.* class MainActivity : Activity() { private lateinit var clock: TextView private lateinit var date: TextView override fun onCreate(savedInstanceState: Bundle?) { super.onCreate(savedInstanceState) window.decorView.systemUiVisibility = View.SYSTEM_UI_FLAG_FULLSCREEN or View.SYSTEM_UI_FLAG_LAYOUT_STABLE createHome() updateClock() Timer().scheduleAtFixedRate(object : TimerTask() { override fun run() { runOnUiThread { updateClock() } } }, 0, 1000) } private fun createHome() { val root = LinearLayout(this) root.orientation = LinearLayout.VERTICAL root.setPadding(20, 20, 20, 20) root.setBackgroundColor(Color.rgb(10, 15, 35)) clock = TextView(this) clock.textSize = 55f clock.setTextColor(Color.WHITE) clock.gravity = Gravity.CENTER root.addView( clock, LinearLayout.LayoutParams( -1, 90 ) ) date = TextView(this) date.textSize = 17f date.setTextColor(Color.WHITE) date.gravity = Gravity.CENTER root.addView( date, LinearLayout.LayoutParams( -1, 45 ) ) val grid = GridLayout(this) grid.columnCount = 4 grid.rowCount = 3 addApp(grid, "📞", "Phone") { openApp("com.google.android.dialer") } addApp(grid, "💬", "Messages") { openApp("com.google.android.apps.messaging") } addApp(grid, "📷", "Camera") { val intent = Intent("android.media.action.IMAGE_CAPTURE") startActivity(intent) } addApp(grid, "⚙️", "Settings") { startActivity( Intent(Settings.ACTION_SETTINGS) ) } addApp(grid, "🌐", "Browser") { val intent = Intent(Intent.ACTION_VIEW) intent.data = android.net.Uri.parse("https://www.google.com") startActivity(intent) } addApp(grid, "📁", "Files") { val intent = Intent(Intent.ACTION_OPEN_DOCUMENT) intent.type = "*/*" intent.addCategory( Intent.CATEGORY_OPENABLE ) startActivity(intent) } addApp(grid, "📅", "Calendar") { startActivity( Intent( Intent.ACTION_VIEW, android.net.Uri.parse( "content://com.android.calendar/time/" ) ) ) } addApp(grid, "🎵", "Music") { val intent = Intent( Intent.ACTION_VIEW, android.net.Uri.parse( "content://media/external/audio/media/" ) ) startActivity(intent) } addApp(grid, "🧭", "Search") { showAppList() } addApp(grid, "🔋", "Battery") { startActivity( Intent(Settings.ACTION_BATTERY_SAVER_SETTINGS) ) } addApp(grid, "📶", "Network") { startActivity( Intent(Settings.ACTION_WIRELESS_SETTINGS) ) } addApp(grid, "ℹ️", "About") { Toast.makeText( this, "JarexOS v1.0", Toast.LENGTH_LONG ).show() } root.addView( grid, LinearLayout.LayoutParams( -1, 0, 1f ) ) val dock = LinearLayout(this) dock.gravity = Gravity.CENTER dock.setPadding(10, 15, 10, 15) addDock(dock, "📞") { openApp("com.google.android.dialer") } addDock(dock, "💬") { openApp("com.google.android.apps.messaging") } addDock(dock, "🌐") { val i = Intent( Intent.ACTION_VIEW, android.net.Uri.parse( "https://www.google.com" ) ) startActivity(i) } addDock(dock, "⚙️") { startActivity( Intent(Settings.ACTION_SETTINGS) ) } root.addView( dock, LinearLayout.LayoutParams( -1, 90 ) ) setContentView(root) } private fun addApp( grid: GridLayout, emoji: String, name: String, action: () -> Unit ) { val box = LinearLayout(this) box.orientation = LinearLayout.VERTICAL box.gravity = Gravity.CENTER box.setPadding(5, 10, 5, 10) val icon = TextView(this) icon.text = emoji icon.textSize = 35f icon.gravity = Gravity.CENTER val label = TextView(this) label.text = name label.textSize = 12f label.setTextColor(Color.WHITE) label.gravity = Gravity.CENTER box.addView(icon) box.addView(label) box.setOnClickListener { action() } val params = GridLayout.LayoutParams() params.width = 0 params.height = 120 params.columnSpec = GridLayout.spec( GridLayout.UNDEFINED, 1f ) grid.addView(box, params) } private fun addDock( dock: LinearLayout, emoji: String, action: () -> Unit ) { val button = TextView(this) button.text = emoji button.textSize = 32f button.gravity = Gravity.CENTER button.setOnClickListener { action() } dock.addView( button, LinearLayout.LayoutParams( 0, -1, 1f ) ) } private fun openApp(packageName: String) { try { val intent = packageManager .getLaunchIntentForPackage( packageName ) if (intent != null) { startActivity(intent) } else { showAppList() } } catch (e: Exception) { showAppList() } } private fun showAppList() { val apps = packageManager .getInstalledApplications(0) .sortedBy { packageManager .getApplicationLabel(it) .toString() } val names = apps.map { packageManager .getApplicationLabel(it) .toString() } AlertDialog.Builder(this) .setTitle("JarexOS Apps") .setItems(names.toTypedArray()) { _, which -> val app = apps[which] val intent = packageManager .getLaunchIntentForPackage( app.packageName ) if (intent != null) startActivity(intent) } .setNegativeButton( "Close", null ) .show() } private fun updateClock() { val now = Date() clock.text = SimpleDateFormat( "h:mm", Locale.getDefault() ).format(now) date.text = SimpleDateFormat( "EEEE, MMMM d", Locale.getDefault() ).format(now) } }
